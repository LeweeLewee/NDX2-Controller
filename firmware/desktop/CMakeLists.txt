cmake_minimum_required(VERSION 3.16)
project(ndx_fixture C)
set(CMAKE_C_STANDARD 99)
if(CMAKE_C_COMPILER MATCHES "zig.exe$")
  set(CMAKE_C_ARCHIVE_CREATE "\"${CMAKE_C_COMPILER}\" ar qc <TARGET> <OBJECTS>")
  set(CMAKE_C_ARCHIVE_FINISH "\"${CMAKE_C_COMPILER}\" ranlib <TARGET>")
endif()
find_package(SDL2 2.0.18 REQUIRED)
if(NOT LVGL_SOURCE_DIR)
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR to checksum-pinned vendor components/lvgl__lvgl")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
add_subdirectory(${LVGL_SOURCE_DIR} lvgl-build)
if(NOT CJSON_SOURCE_DIR)
  message(FATAL_ERROR "Set CJSON_SOURCE_DIR to pinned ESP-IDF components/json/cJSON")
endif()
add_library(bridge_protocol STATIC ../shared/bridge_protocol.c ${CJSON_SOURCE_DIR}/cJSON.c)
target_compile_definitions(bridge_protocol PRIVATE CJSON_NESTING_LIMIT=16)
target_include_directories(bridge_protocol PUBLIC ../shared ${CJSON_SOURCE_DIR})
add_executable(ndx_fixture main.c transport.c ../shared/controller.c ../shared/ui.c ../shared/fixture.c)
target_include_directories(ndx_fixture PRIVATE ../shared ${SDL2_INCLUDE_DIRS})
target_link_libraries(ndx_fixture PRIVATE lvgl bridge_protocol SDL2::SDL2 bcrypt)
add_executable(controller_test ../../tests/controller_test.c ../shared/controller.c)
target_include_directories(controller_test PRIVATE ../shared)
enable_testing()
add_test(NAME controller_state COMMAND controller_test)
add_executable(bridge_protocol_test ../../tests/bridge_protocol_test.c)
target_link_libraries(bridge_protocol_test PRIVATE bridge_protocol)
add_test(NAME bridge_protocol COMMAND bridge_protocol_test)
